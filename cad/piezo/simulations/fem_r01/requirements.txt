# FEM pipeline dependencies. These live in `pic-env` (3.11), NOT in the `cad`
# environment: the cad env carries OCP, whose teardown segfaults on this
# machine, and it has no meshing stack.
#
# Deliberately separate from ../../requirements.txt, which is an input to the
# P0 Fusion build manifest - editing that file invalidates `build.py --check`.
#
# pypardiso is not optional in practice: scipy's SuperLU solved 104k dofs in
# 49 s and then failed to finish 188k dofs in 24 minutes, while MKL PARDISO
# does 188k in 88 s and 1.2M in 709 s. The code falls back to scipy without it,
# but only the coarsest densities are then reachable.
gmsh==4.12.2
scikit-fem==10.0.2
meshio==5.3.5
pypardiso==0.4.7
matplotlib==3.10.9

# Installed as a fallback preconditioner in case the direct solver ran out of
# memory at the finest mesh. It did not, so nothing imports pyamg today.
pyamg==5.3.0
